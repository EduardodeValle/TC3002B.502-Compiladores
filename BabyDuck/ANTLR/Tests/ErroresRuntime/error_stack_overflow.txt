vars
    resultado : entero;

    {
        escribe(n);
        devolver recursionInfinita(n + 1);
    }
};

    {
        escribe("Iniciando recursion infinita...");
        resultado = recursionInfinita(1);
        escribe("Este mensaje nunca se imprimira");
        escribe(resultado);
    }
fin
